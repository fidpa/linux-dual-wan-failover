# Runtime dependencies for the linux-dual-wan-failover Web-UI.
#
# Install into a per-deployment venv via `install.sh --with-web-ui` (which
# creates /usr/local/lib/linux-dual-wan-failover/web/venv and pip-installs
# this file). For local development:
#
#     python3 -m venv .venv
#     .venv/bin/pip install -r src/web/requirements.txt
#
# Versions are pinned to the latest stable line on the 3.x branch as of
# the v0.2.0 release. Re-pin during a security audit; nothing here has
# unstable APIs.

Flask>=3.1,<4.0
gunicorn>=23.0,<26.0
Jinja2>=3.1,<4.0       # Flask brings this in transitively; pinned for SLSA hashing
Werkzeug>=3.1,<4.0     # likewise transitive

# Test-only (kept here for `pip install -r requirements.txt` simplicity in
# CI). Production installs should use `pip install --no-deps` or split this
# into a separate dev-requirements.txt if the extra footprint matters.
pytest>=8.0,<9.0
